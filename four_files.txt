    88	        ReadOnlySpan<byte> utf8Json,
    89	        out LiveStreamFrame frame,
    90	        out double[]? ohlcv)
    91	    {
    92	        ohlcv = null;
    93	        if (!TryParseBinanceKline(utf8Json, out frame))
    94	            return false;
    95	
    96	        try
    97	        {
    98	            using var doc = JsonDocument.Parse(utf8Json.ToArray());
    99	            var root = doc.RootElement;
   100	            var payload = root.TryGetProperty("data", out var dataElem) ? dataElem : root;
   101	            if (payload.TryGetProperty("k", out var kline))
   102	            {
   103	                double c = (double)frame.Price;
   104	                double o = kline.TryGetProperty("o", out var op) && double.TryParse(op.GetString(), NumberStyles.Any, CultureInfo.InvariantCulture, out var ov) ? ov : c;
   105	                double h = kline.TryGetProperty("h", out var hp) && double.TryParse(hp.GetString(), NumberStyles.Any, CultureInfo.InvariantCulture, out var hv) ? hv : Math.Max(o, c);
   106	                double l = kline.TryGetProperty("l", out var lp) && double.TryParse(lp.GetString(), NumberStyles.Any, CultureInfo.InvariantCulture, out var lv) ? lv : Math.Min(o, c);
   107	                double v = kline.TryGetProperty("v", out var vp) && double.TryParse(vp.GetString(), NumberStyles.Any, CultureInfo.InvariantCulture, out var vv) ? vv : 1.0;
   108	                ohlcv = new[] { o, h, l, c, v };
   109	            }
   110	        }
   111	        catch
   112	        {
   113	            // Keep frame even if optional OHLCV fields are absent
   114	        }
   115	        return true;

   119	    {
   120	        frame = default;

   122	        {
   123	            using var doc = JsonDocument.Parse(utf8Json.ToArray());

   148	    private async Task RunSyntheticGbmFeedAsync(CancellationToken ct)
   149	    {
   150	        var prices = new Dictionary<string, decimal>
   151	        {
   152	            ["BTCUSDT"] = 88500.00m,
   153	            ["ETHUSDT"] = 3150.00m,
   154	            ["SOLUSDT"] = 185.00m
   155	        };
   156	        var rng = new Random();
   157	
   158	        while (!ct.IsCancellationRequested)
   159	        {
   160	            foreach (var sym in prices.Keys.ToArray())
   161	            {
   162	                // Box-Muller normal shock: drift + volatility
   163	                double u1 = 1.0 - rng.NextDouble();
   164	                double u2 = 1.0 - rng.NextDouble();
   165	                double z = Math.Sqrt(-2.0 * Math.Log(u1)) * Math.Sin(2.0 * Math.PI * u2);
   166	                decimal pctMove = (decimal)(0.00005 + 0.0012 * z);
   167	
   168	                prices[sym] = Math.Round(Math.Max(0.01m, prices[sym] * (1.0m + pctMove)), 4);
   169	                _gateway.PublishFrame(new LiveStreamFrame(
   170	                    "MARKET_TICK",
   171	                    sym,
   172	                    prices[sym],
   173	                    null,
   174	                    "SYNTHETIC_GBM",
   175	                    DateTimeOffset.UtcNow.ToUnixTimeMilliseconds()));
   176	            }
   177	            try
   178	            {
   179	                await Task.Delay(250, ct);
   180	            }
   181	            catch (OperationCanceledException)
   182	            {
   183	                break;
   184	            }
   185	        }

   119	    {
   120	        var candidateSymbols = JsonSerializer.Deserialize<List<string>>(slot.CandidateSymbolsJson)
   121	            ?? new List<string> { "SOLUSDT" };
   122	        var strat = ParseStrategyPayload(slot.StrategyPayloadJson);
   123	
   124	        bool isCarryMode = _settlementStrategy.StrategyName.Equals("CARRY_POSITIONS", StringComparison.OrdinalIgnoreCase);
   125	
   126	        var openQuantities = new Dictionary<string, decimal>();
   127	        var entryPrices = new Dictionary<string, decimal>();
   128	        var latestPrices = new Dictionary<string, decimal>();
   129	        var moves = new List<ExecutedMoveRecord>();
   130	
   131	        using var durationCts = new CancellationTokenSource(TimeSpan.FromSeconds(maxSeconds));
   132	        using var linkedCts = CancellationTokenSource.CreateLinkedTokenSource(
   133	            stoppingToken,
   134	            durationCts.Token,
   135	            _commandWatcher.CurrentEpisodeResetToken);
   136	
   137	        int ticksProcessed = 0;
   138	        decimal realizedPnlUsdt = 0m;
   139	        decimal initialNotionalUsdt = 0m;
   140	        decimal peakEquityUsdt = 0m;
   141	        decimal prevEquityUsdt = 0m;
   142	        decimal maxDrawdownPct = 0m;
   143	
   144	        // Welford online variance accumulators for tick-by-tick returns
   145	        int returnCount = 0;
   146	        double meanReturn = 0.0;
   147	        double m2Return = 0.0;
   148	
   149	        try
   150	        {
   151	            while (ticksProcessed < maxTicks && !linkedCts.Token.IsCancellationRequested)
   152	            {
   153	                var tick = await _commandWatcher.MarketTicksReader.ReadAsync(linkedCts.Token);
   154	                if (tick.Price <= 0m) continue;
   155	
   156	                latestPrices[tick.Symbol] = tick.Price;
   157	                ticksProcessed++;
   158	
   159	                // 1. Entry or Carried Position Adoption
   160	                if (openQuantities.Count == 0 && entryPrices.Count == 0 && candidateSymbols.Contains(tick.Symbol))
   161	                {
   162	                    string carryKey = $"{slot.ProfileKey}:{tick.Symbol}";
   163	
   164	                    if (isCarryMode && _carriedPositions.TryRemove(carryKey, out var carried) && carried.Quantity > 0m)
   165	                    {
   166	                        // Re-lock slice in API mutex without paying a duplicate 10 bps entry taker fee
   167	                        var claim = await _apiClient.ClaimSymbolSliceAsync(
   168	                            new ShareSliceClaimRequest(
   169	                                agentId,
   170	                                slot.PoolInstanceId,
   171	                                tick.Symbol,
   172	                                strat.MaxSliceClaimPct,
   173	                                "LONG_ACCUMULATING"),
   174	                            linkedCts.Token);
   175	
   176	                        decimal activeQty = (claim is { Claimed: true, ReservedQuantity: > 0m })
   177	                            ? claim.ReservedQuantity
   178	                            : carried.Quantity;
   179	
   180	                        // Only charge taker fee on net incremental quantity added above the carried position
   181	                        decimal incrementalQty = Math.Max(0m, activeQty - carried.Quantity);
   182	                        decimal deltaFee = Math.Round(incrementalQty * tick.Price * MakerFeeRate, 8);
   183	
   184	                        // Mark-to-market baseline from current tick price for this episode's delta PnL
   185	                        openQuantities[tick.Symbol] = activeQty;
   186	                        entryPrices[tick.Symbol] = tick.Price;
   187	                        realizedPnlUsdt -= deltaFee;
   188	
   189	                        initialNotionalUsdt = activeQty * tick.Price;
   190	                        peakEquityUsdt = initialNotionalUsdt - deltaFee;
   191	                        prevEquityUsdt = peakEquityUsdt;
   192	
   193	                        if (incrementalQty > 0m)
   194	                        {
   195	                            var orderRes = await _executionGateway.SubmitOrderAsync(
   196	                                new ExecutionOrderRequest(Guid.NewGuid(), tick.Symbol, "buy", incrementalQty, tick.Price, strat.IsPromotedLive, slot.ReplicaIndex, strat.ModelConfidence, strat.ExpectedEdgeBps, "ENTRY", slot.ProfileKey),
   197	                                linkedCts.Token);
   198	                            if (!orderRes.Accepted)
   199	                            {
   200	                                _logger.LogWarning("[Agent {AgentId}] Carry delta order rejected ({Provider}/{Mode}): {Msg}",
   201	                                    agentId, orderRes.Provider, orderRes.Mode, orderRes.Message);
   202	                            }
   203	                        }
   204	
   205	                        moves.Add(new ExecutedMoveRecord(
   206	                            tick.Symbol,
   207	                            "CARRY_INHERIT_SLICE",
   208	                            activeQty,
   209	                            tick.Price,
   210	                            deltaFee,
   211	                            tick.TimestampMs));
   212	                    }
   213	                    else
   214	                    {
   215	                        var claim = await _apiClient.ClaimSymbolSliceAsync(
   216	                            new ShareSliceClaimRequest(
   217	                                agentId,
   218	                                slot.PoolInstanceId,
   219	                                tick.Symbol,
   220	                                strat.MaxSliceClaimPct,
   221	                                "LONG_ACCUMULATING"),
   222	                            linkedCts.Token);
   223	
   224	                        if (claim is { Claimed: true, ReservedQuantity: > 0m })
   225	                        {
   226	                            decimal entryFee = Math.Round(claim.ReservedQuantity * tick.Price * MakerFeeRate, 8);
   227	                            openQuantities[tick.Symbol] = claim.ReservedQuantity;
   228	                            entryPrices[tick.Symbol] = tick.Price;
   229	                            realizedPnlUsdt -= entryFee;
   230	
   231	                            initialNotionalUsdt = claim.ReservedQuantity * tick.Price;
   232	                            peakEquityUsdt = initialNotionalUsdt - entryFee;
   233	                            prevEquityUsdt = peakEquityUsdt;
   234	
   235	                            var orderRes = await _executionGateway.SubmitOrderAsync(
   236	                                new ExecutionOrderRequest(Guid.NewGuid(), tick.Symbol, "buy", claim.ReservedQuantity, tick.Price, strat.IsPromotedLive, slot.ReplicaIndex, strat.ModelConfidence, strat.ExpectedEdgeBps, "ENTRY", slot.ProfileKey),
   237	                                linkedCts.Token);
   238	                            if (!orderRes.Accepted)
   239	                            {
   240	                                _logger.LogWarning("[Agent {AgentId}] Entry order rejected ({Provider}/{Mode}): {Msg}",
   241	                                    agentId, orderRes.Provider, orderRes.Mode, orderRes.Message);
   242	                            }
   243	
   244	                            moves.Add(new ExecutedMoveRecord(
   245	                                tick.Symbol,
   246	                                "BUY_SLICE_CLAIM",
   247	                                claim.ReservedQuantity,
   248	                                tick.Price,
   249	                                entryFee,
   250	                                tick.TimestampMs));
   251	                        }
   252	                    }
   253	                    continue;
   254	                }
   255	
   256	                // 2. Mark-to-Market Valuation, Stop-Loss / Take-Profit Check & Online Sharpe Update
   257	                if (openQuantities.TryGetValue(tick.Symbol, out decimal heldQty) && heldQty > 0m)
   258	                {
   259	                    decimal entryPrice = entryPrices[tick.Symbol];
   260	                    decimal priceReturnPct = (tick.Price - entryPrice) / entryPrice;
   261	
   262	                    decimal stopLossPct = -strat.StopLossAtr * BaseAtrPct;
   263	                    decimal takeProfitPct = strat.StopLossAtr * BaseAtrPct * strat.TakeProfitRr;
   264	
   265	                    decimal currentPositionValue = heldQty * tick.Price;
   266	                    decimal currentEquity = initialNotionalUsdt + realizedPnlUsdt + (heldQty * (tick.Price - entryPrice));
   267	
   268	                    if (prevEquityUsdt > 0m)
   269	                    {
   270	                        double stepRet = (double)((currentEquity - prevEquityUsdt) / prevEquityUsdt);
   271	                        returnCount++;
   272	                        double delta = stepRet - meanReturn;
   273	                        meanReturn += delta / returnCount;
   274	                        double delta2 = stepRet - meanReturn;
   275	                        m2Return += delta * delta2;
   276	                    }
   277	                    prevEquityUsdt = currentEquity;
   278	
   279	                    if (currentEquity > peakEquityUsdt)
   280	                    {
   281	                        peakEquityUsdt = currentEquity;
   282	                    }
   283	                    else if (peakEquityUsdt > 0m)
   284	                    {
   285	                        decimal dd = (peakEquityUsdt - currentEquity) / peakEquityUsdt;
   286	                        if (dd > maxDrawdownPct) maxDrawdownPct = Math.Round(dd, 4);
   287	                    }
   288	
   289	                    // Dynamic Stop-Loss or Take-Profit Exit
   290	                    if (priceReturnPct <= stopLossPct || priceReturnPct >= takeProfitPct)
   291	                    {
   292	                        string exitAction = priceReturnPct >= takeProfitPct ? "SELL_TAKE_PROFIT" : "SELL_STOP_LOSS";
   293	                        decimal exitFee = Math.Round(currentPositionValue * (exitAction == "SELL_TAKE_PROFIT" ? MakerFeeRate : TakerFeeRate), 8);
   294	                        decimal tradeGrossPnl = heldQty * (tick.Price - entryPrice);
   295	
   296	                        realizedPnlUsdt += tradeGrossPnl - exitFee;
   297	                        openQuantities[tick.Symbol] = 0m;
   298	
   299	                        var sellRes = await _executionGateway.SubmitOrderAsync(
   300	                            new ExecutionOrderRequest(Guid.NewGuid(), tick.Symbol, "sell", heldQty, tick.Price, strat.IsPromotedLive, slot.ReplicaIndex, strat.ModelConfidence, strat.ExpectedEdgeBps, exitAction == "SELL_TAKE_PROFIT" ? "TAKE_PROFIT" : "STOP_LOSS", slot.ProfileKey),
   301	                            linkedCts.Token);
   302	                        if (!sellRes.Accepted)
   303	                        {
   304	                            _logger.LogWarning("[Agent {AgentId}] Exit order rejected ({Provider}/{Mode}): {Msg}",
   305	                                agentId, sellRes.Provider, sellRes.Mode, sellRes.Message);
   306	                        }
   307	
   308	                        moves.Add(new ExecutedMoveRecord(
   309	                            tick.Symbol,
   310	                            exitAction,
   311	                            heldQty,
   312	                            tick.Price,
   313	                            exitFee,
   314	                            tick.TimestampMs));
   315	                        break; // Position closed; settle episode immediately
   316	                    }
   317	                }
   318	            }
   319	        }
   320	        catch (OperationCanceledException) when (!stoppingToken.IsCancellationRequested)
   321	        {
   322	            _logger.LogInformation("[Agent {AgentId}] Episode interrupted by Precedence Reset or Time Window.", agentId);
   323	        }
   324	
   325	        // Evaluate open positions at episode boundary
   326	        foreach (var (sym, qty) in openQuantities)
   327	        {
   328	            if (qty > 0m && latestPrices.TryGetValue(sym, out decimal lastP) && entryPrices.TryGetValue(sym, out decimal entP))
   329	            {
   330	                realizedPnlUsdt += qty * (lastP - entP);
   331	
   332	                if (isCarryMode)
   333	                {
   334	                    _carriedPositions[$"{slot.ProfileKey}:{sym}"] = new CarriedPositionState(sym, qty, lastP);
   335	                }
   336	            }
   337	        }
   338	
   339	        decimal sharpeEstimate = ComputeAnnualizedSharpe(returnCount, meanReturn, m2Return);
   340	
   341	        var context = new ActiveAgentContext(
   342	            RunId: Guid.NewGuid(),
   343	            PoolInstanceId: slot.PoolInstanceId,
   344	            AgentId: agentId,
   345	            ActorType: "AGENT",
   346	            ProfileKey: slot.ProfileKey,
   347	            ReplicaIndex: slot.ReplicaIndex,
   348	            JitterDeltaJson: slot.JitterDeltaJson,
   349	            ExecutedMoves: moves,
   350	            RealizedPnlUsdt: Math.Round(realizedPnlUsdt, 8),
   351	            MaxDrawdownPct: Math.Round(maxDrawdownPct, 4),
   352	            SharpeEstimate: sharpeEstimate,
   353	            OpenSymbolQuantities: openQuantities,
   354	            LatestMarketPrices: latestPrices
   355	        );
   356	
   357	        return await _settlementStrategy.SettleEpisodeAsync(context, stoppingToken);
   358	    }

   360	    public static ParsedStrategyParams ParseStrategyPayload(string json)
   361	    {
   362	        try
   363	        {
   364	            using var doc = JsonDocument.Parse(json);
   365	            var root = doc.RootElement;
   366	            decimal entry = root.TryGetProperty("entry_threshold", out var e) ? e.GetDecimal() : 0.60m;
   367	            decimal sl = root.TryGetProperty("stop_loss_atr", out var s) ? s.GetDecimal() : 1.50m;
   368	            decimal tp = root.TryGetProperty("take_profit_rr", out var t) ? t.GetDecimal() : 2.00m;
   369	            decimal slice = root.TryGetProperty("max_slice_claim_pct", out var m) ? m.GetDecimal() : 0.25m;
   370	            bool isPromoted = root.TryGetProperty("is_promoted_live", out var pr) && pr.ValueKind == JsonValueKind.True;
   371	            decimal modelConf = root.TryGetProperty("confidence_score", out var cf) && cf.TryGetDecimal(out var cVal) ? cVal : 0m;
   372	            decimal edgeBps = root.TryGetProperty("expected_return_bps", out var eb) && eb.TryGetDecimal(out var eVal) ? eVal : 0m;
   373	            return new ParsedStrategyParams(entry, sl, tp, Math.Clamp(slice, 0.01m, 1.0m), isPromoted, modelConf, edgeBps);
   374	        }
   375	        catch
   376	        {
   377	            return new ParsedStrategyParams(0.60m, 1.50m, 2.00m, 0.25m);
   378	        }
   379	    }
   380	
   381	    public static decimal ComputeAnnualizedSharpe(int count, double mean, double m2)
   382	    {
   383	        if (count < 2) return 0m;
   384	        double variance = m2 / (count - 1);
   385	        double stdDev = Math.Sqrt(Math.Max(variance, 1e-12));
   386	        if (stdDev <= 1e-9) return 0m;
   387	        double rawSharpe = (mean / stdDev) * Math.Sqrt(count);
   388	        return Math.Round((decimal)Math.Clamp(rawSharpe, -10.0, 10.0), 4);
   389	    }
