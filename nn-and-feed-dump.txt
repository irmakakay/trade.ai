import json
import os
from uuid import uuid4
import pika
import psycopg
from psycopg.rows import dict_row

from adapter.queue_consumer import INPUT_QUEUE, OUTPUT_QUEUE, process_single_task
from adapter.schemas import InputTaskMessage, TaskCommandType
from quant_nn_engine import BrainStateCodec, QuantBrainBridge, ReplicaHistoryRecord

CONFIDENCE_THRESHOLD = float(os.getenv("CONFIDENCE_THRESHOLD", "0.75"))
BRIDGE = QuantBrainBridge(confidence_threshold=CONFIDENCE_THRESHOLD, hidden_dim=64)


def get_pg_conn_str() -> str:
    return (
        f"host={os.getenv('POSTGRES_HOST', 'localhost')} "
        f"port={os.getenv('POSTGRES_PORT', '5432')} "
        f"dbname={os.getenv('POSTGRES_DB', 'quantcore')} "
        f"user={os.getenv('POSTGRES_USER', 'quant')} "
        f"password={os.getenv('POSTGRES_PASSWORD', 'quant_secret')}"
    )


def fetch_latest_generation_histories(cur, profile_key: str, limit: int) -> list[ReplicaHistoryRecord]:
    cur.execute(
        """
        SELECT replica_index, jitter_delta, realized_pnl_usdt, max_drawdown_pct, sharpe_estimate
        FROM trading_history
        WHERE profile_key = %s
        ORDER BY completed_at DESC
        LIMIT %s;
        """,
        (profile_key, limit),
    )
    rows = cur.fetchall()
    return [
        ReplicaHistoryRecord(
            replica_index=int(r["replica_index"]),
            jitter_delta=r["jitter_delta"] if isinstance(r["jitter_delta"], dict) else json.loads(r["jitter_delta"]),
            realized_pnl_usdt=float(r["realized_pnl_usdt"]),
            max_drawdown_pct=float(r["max_drawdown_pct"]),
            sharpe_estimate=float(r["sharpe_estimate"]),
        )
        for r in rows
    ]


def handle_task(task: InputTaskMessage) -> tuple[float, dict]:
    with psycopg.connect(get_pg_conn_str(), row_factory=dict_row) as conn:
        with conn.cursor() as cur:
            prior_histories = fetch_latest_generation_histories(
                cur, task.profile_key, task.replication_factor
            )

            confidence, result = BRIDGE.calculate_pool_and_forecast(
                profile_key=task.profile_key,
                risk_dial_r=task.risk_dial_r,
                candidate_symbols=task.candidate_symbols,
                replication_factor=task.replication_factor,
                prior_histories=prior_histories,
            )

            if task.command == TaskCommandType.RECALCULATE_POOL:
                # Clear stale AVAILABLE slots for this profile before inserting the new generation
                cur.execute(
                    "DELETE FROM strategy_pool WHERE profile_key = %s AND status = 'AVAILABLE';",
                    (task.profile_key,),
                )
                for rep in result["replicas"]:
                    cur.execute(
                        """
                        INSERT INTO strategy_pool
                            (pool_instance_id, profile_key, replica_index, risk_dial_r, jitter_delta, strategy_payload, candidate_symbols, status)
                        VALUES (%s, %s, %s, %s, %s::jsonb, %s::jsonb, %s::jsonb, 'AVAILABLE');
                        """,
                        (
                            str(uuid4()),
                            task.profile_key,
                            rep["replica_index"],
                            task.risk_dial_r,
                            json.dumps(rep["jitter_delta"]),
                            json.dumps(rep["strategy_payload"]),
                            json.dumps(rep["candidate_symbols"]),
                        ),
                    )

                # Persist BYTEA model checkpoint
                raw_bundle = BrainStateCodec.dump_state_bundle(
                    BRIDGE.model,
                    BRIDGE.optimizer,
                    {"profile_key": task.profile_key},
                    confidence,
                )
                cur.execute(
                    """
                    INSERT INTO nn_model_checkpoints
                        (model_id, version_tag, architecture_config, weights_blob, optimizer_blob, scaler_blob, confidence_score, is_promoted_live)
                    VALUES (%s, %s, %s::jsonb, %s, %s, %s, %s, %s);
                    """,
                    (
                        str(uuid4()),
                        f"{task.profile_key}_gen{result.get('generation', 1)}",
                        json.dumps({"hidden_dim": BRIDGE.model.hidden_dim}),
                        raw_bundle,
                        b"",
                        b"{}",
                        confidence,
                        result["is_promoted_for_live"],
                    ),
                )
        conn.commit()
    print(
        f"[NN-EVOLUTION] Profile={task.profile_key} | Gen={result.get('generation', 0)} | "
        f"Winner=Replica#{result['winning_prior_replica']} | Loss={result.get('train_loss', 0.0):.4f} | "
        f"Confidence={confidence*100:.1f}% | Promoted={result['is_promoted_for_live']}",
        flush=True,
    )

    return confidence, {
        "profile_key": task.profile_key,
        "winning_prior_replica": result["winning_prior_replica"],
        "is_promoted_for_live": result["is_promoted_for_live"],
        "replicas_generated": len(result["replicas"]),
    }


def main() -> None:
    rmq_host = os.getenv("RABBITMQ_HOST", "localhost")
    rmq_port = int(os.getenv("RABBITMQ_PORT", "5672"))
    params = pika.ConnectionParameters(host=rmq_host, port=rmq_port)
    connection = pika.BlockingConnection(params)
    channel = connection.channel()

    channel.queue_declare(queue=INPUT_QUEUE, durable=True)
    channel.queue_declare(
        queue=OUTPUT_QUEUE,
        durable=True,
        arguments={"x-max-length": 1, "x-overflow": "drop-head"},
    )
    channel.basic_qos(prefetch_count=1)

    def _on_message(ch, method, properties, body):
        try:
            process_single_task(ch, body, handle_task)
        finally:
            ch.basic_ack(delivery_tag=method.delivery_tag)

    channel.basic_consume(queue=INPUT_QUEUE, on_message_callback=_on_message)
    print("[quant_nn_worker] Connected to quant_nn_engine. Listening on input_queue...")
    channel.start_consuming()


if __name__ == "__main__":
    main()

from typing import Dict, List, Optional, Tuple
import numpy as np
import torch
from contracts import (
    AssetMoveForecast,
    ForecastPackage,
    ReplicaHistoryRecord,
    StrategyHyperparams,
    TradeActionType,
)
from data.indicators import compute_canonical_pattern_priors
from evolution.empirical_gradient import compute_replica_fitness, evolve_base_strategy_from_histories
from evolution.jitter_pool import build_antithetic_jitter_pool
from models.loss import DifferentiableNavSharpeLoss
from models.network import PortfolioConditionedQuantNet


class QuantBrainBridge:
    """
    High-level facade connecting `quant_nn_worker` task loop to `quant_nn_engine`.
    Performs online gradient updates on completed K-replica episodes during RECALCULATE_POOL.
    """
    def __init__(self, confidence_threshold: float = 0.75, hidden_dim: int = 64):
        self.confidence_threshold = confidence_threshold
        self.model = PortfolioConditionedQuantNet(hidden_dim=hidden_dim)
        self.optimizer = torch.optim.AdamW(self.model.parameters(), lr=4e-3, weight_decay=1e-4)
        self.loss_fn = DifferentiableNavSharpeLoss()
        self.generation_count: int = 0
        self.generation_by_profile: Dict[str, int] = {}

    def train_step_on_candles(
        self,
        ohlcv_window: np.ndarray,
        portfolio_vec: np.ndarray,
        risk_dial_r: float,
        future_return: float,
    ) -> Tuple[float, float]:
        self.model.train()
        priors = compute_canonical_pattern_priors(ohlcv_window)
        m_tensor = torch.tensor(priors, dtype=torch.float32).unsqueeze(0)
        p_tensor = torch.tensor(portfolio_vec, dtype=torch.float32).unsqueeze(0)
        r_tensor = torch.tensor([[risk_dial_r]], dtype=torch.float32)
        fut_tensor = torch.tensor([future_return], dtype=torch.float32)

        self.optimizer.zero_grad()
        out = self.model(m_tensor, p_tensor, r_tensor)
        loss = self.loss_fn(out, fut_tensor)
        loss.backward()
        self.optimizer.step()

        return float(loss.item()), float(out["confidence_score"].item())

    def train_on_generation_histories(
        self,
        histories: List[ReplicaHistoryRecord],
        risk_dial_r: float,
        profile_key: str = "DEFAULT",
        epochs: int = 6,
    ) -> float:
        """
        Runs online PyTorch gradient updates over the completed K-replica batch so the network's
        strategy head, action head, and confidence gate actively learn from live agent outcomes.
        """
        if not histories:
            return 0.0

        self.model.train()
        self.generation_count += 1
        gen_num = self.generation_by_profile.get(profile_key, 0) + 1
        self.generation_by_profile[profile_key] = gen_num
        batch_size = len(histories)

        # Build informative state tensors encoding each replica's empirical metrics
        m_batch = np.zeros((batch_size, 32, 10), dtype=np.float32)
        p_batch = np.zeros((batch_size, 6), dtype=np.float32)
        returns_list = []
        conf_targets = []

        scored = [compute_replica_fitness(h) for h in histories]
        best_fit = max(scored)

        for i, h in enumerate(histories):
            norm_pnl = float(np.tanh(h.realized_pnl_usdt / 25.0))
            sharpe_clipped = float(np.clip(h.sharpe_estimate / 3.0, -1.0, 1.0))
            m_batch[i, :, 0] = norm_pnl
            m_batch[i, :, 1] = sharpe_clipped
            m_batch[i, :, 2] = float(h.max_drawdown_pct)

            p_batch[i] = [
                0.5,
                0.5,
                float(h.max_drawdown_pct),
                float(np.clip(0.5 + norm_pnl * 0.3, 0.0, 1.0)),
                1.0,
                risk_dial_r,
            ]
            # Effective directional return signal from replica fitness
            eff_ret = norm_pnl if abs(norm_pnl) > 1e-4 else (0.015 if h.sharpe_estimate >= 0 else -0.015)
            returns_list.append(eff_ret)

            # Progressive confidence curriculum as generations accumulate + positive Sharpe
            maturity_bonus = min(0.28, gen_num * 0.055)
            sharpe_bonus = max(0.0, min(0.15, h.sharpe_estimate * 0.06))
            conf_targets.append(min(0.92, 0.52 + maturity_bonus + sharpe_bonus))

        m_tensor = torch.tensor(m_batch, dtype=torch.float32)
        p_tensor = torch.tensor(p_batch, dtype=torch.float32)
        r_tensor = torch.full((batch_size, 1), risk_dial_r, dtype=torch.float32)
        fut_tensor = torch.tensor(returns_list, dtype=torch.float32)
        conf_target_tensor = torch.tensor(conf_targets, dtype=torch.float32)

        last_loss = 0.0
        for _ in range(epochs):
            self.optimizer.zero_grad()
            out = self.model(m_tensor, p_tensor, r_tensor)
            loss = self.loss_fn(out, fut_tensor, target_confidence=conf_target_tensor)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(self.model.parameters(), 1.0)
            self.optimizer.step()
            last_loss = float(loss.item())

        return last_loss

    def calculate_pool_and_forecast(
        self,
        profile_key: str,
        risk_dial_r: float,
        candidate_symbols: List[str],
        replication_factor: int,
        ohlcv_by_symbol: Optional[Dict[str, np.ndarray]] = None,
        prior_histories: Optional[List[ReplicaHistoryRecord]] = None,
    ) -> Tuple[float, Dict]:
        # 1. If prior generation histories exist, run online PyTorch training first!
        train_loss = 0.0
        if prior_histories:
            train_loss = self.train_on_generation_histories(prior_histories, risk_dial_r, profile_key=profile_key)

        # 2. Switch to eval mode to synthesize updated strategy parameters & confidence
        self.model.eval()
        p_tensor = torch.tensor([[0.5, 0.5, 0.0, 0.55, 1.0, risk_dial_r]], dtype=torch.float32)
        r_tensor = torch.tensor([[risk_dial_r]], dtype=torch.float32)

        asset_forecasts: List[AssetMoveForecast] = []
        base_params_accum = []

        for sym in candidate_symbols:
            if ohlcv_by_symbol and sym in ohlcv_by_symbol:
                priors = compute_canonical_pattern_priors(ohlcv_by_symbol[sym])
            else:
                priors = np.zeros((32, 10), dtype=np.float32)

            m_tensor = torch.tensor(priors, dtype=torch.float32).unsqueeze(0)
            with torch.no_grad():
                out = self.model(m_tensor, p_tensor, r_tensor)

            probs = torch.softmax(out["action_logits"][0], dim=-1).tolist()
            action_idx = int(np.argmax(probs))
            action = [TradeActionType.BUY, TradeActionType.SELL, TradeActionType.HOLD][action_idx]

            asset_forecasts.append(
                AssetMoveForecast(
                    symbol=sym,
                    suitability_score=round(float(out["suitability_score"][0].item()), 4),
                    recommended_action=action,
                    target_slice_pct=round(float(out["target_slice_pct"][0].item()), 4),
                    expected_return_bps=round((probs[0] - probs[1]) * 45.0, 2),
                )
            )
            base_params_accum.append({
                "entry_threshold": float(out["entry_threshold"][0].item()),
                "stop_loss_atr": float(out["stop_loss_atr"][0].item()),
                "take_profit_rr": float(out["take_profit_rr"][0].item()),
                "max_slice_claim_pct": float(out["max_slice_claim_pct"][0].item()),
                "confidence": float(out["confidence_score"][0].item()),
            })

        avg_base = {
            k: round(float(np.mean([b[k] for b in base_params_accum])), 4)
            for k in ("entry_threshold", "stop_loss_atr", "take_profit_rr", "max_slice_claim_pct")
        }
        confidence = round(float(np.mean([b["confidence"] for b in base_params_accum])), 4)

        winner_replica = 1
        if prior_histories:
            avg_base, winner_replica = evolve_base_strategy_from_histories(avg_base, prior_histories)

        asset_forecasts.sort(key=lambda f: f.suitability_score, reverse=True)
        ranked_symbols = [f.symbol for f in asset_forecasts]

        replicas = build_antithetic_jitter_pool(
            base_strategy=avg_base,
            ranked_symbols=ranked_symbols,
            replication_factor=replication_factor,
        )

        forecast_pkg = ForecastPackage(
            profile_key=profile_key,
            risk_dial_r=risk_dial_r,
            confidence_score=confidence,
            is_promoted_for_live=(confidence >= self.confidence_threshold and (not prior_histories or max(h.sharpe_estimate for h in prior_histories) > 0.0)),
            strategy_params=StrategyHyperparams(**avg_base),
            ranked_symbols=ranked_symbols,
            asset_forecasts=asset_forecasts,
        )

        return confidence, {
            "profile_key": profile_key,
            "generation": self.generation_by_profile.get(profile_key, self.generation_count),
            "train_loss": round(train_loss, 4),
            "winning_prior_replica": winner_replica,
            "is_promoted_for_live": forecast_pkg.is_promoted_for_live,
            "replicas": replicas,
            "forecast": forecast_pkg.model_dump(),
        }

using System.Globalization;
using System.Net.WebSockets;
using System.Text;
using System.Text.Json;
using QuantCore.Contracts.Domain;

namespace QuantCore.Api.Services;

public sealed class BinanceMarketFeedHostedService : BackgroundService
{
    private readonly MarketStreamGateway _gateway;
    private readonly IConfiguration _config;
    private readonly ILogger<BinanceMarketFeedHostedService> _logger;

    public BinanceMarketFeedHostedService(
        MarketStreamGateway gateway,
        IConfiguration config,
        ILogger<BinanceMarketFeedHostedService> logger)
    {
        _gateway = gateway;
        _config = config;
        _logger = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        bool forceSynthetic = _config.GetValue<bool>("USE_SYNTHETIC_MARKET_FEED", false);
        string wsUrl = _config["BINANCE_WS_URL"]
            ?? "wss://stream.binance.com:9443/stream?streams=btcusdt@kline_1m/ethusdt@kline_1m/solusdt@kline_1m";

        var buffer = new byte[8192];

        while (!stoppingToken.IsCancellationRequested)
        {
            if (forceSynthetic)
            {
                await RunSyntheticGbmFeedAsync(stoppingToken);
                continue;
            }

            try
            {
                using var ws = new ClientWebSocket();
                using var connectCts = CancellationTokenSource.CreateLinkedTokenSource(stoppingToken);
                connectCts.CancelAfter(TimeSpan.FromSeconds(5));

                await ws.ConnectAsync(new Uri(wsUrl), connectCts.Token);
                _logger.LogInformation("[BinanceMarketFeed] Connected to live stream: {Url}", wsUrl);

                while (ws.State == WebSocketState.Open && !stoppingToken.IsCancellationRequested)
                {
                    var res = await ws.ReceiveAsync(buffer, stoppingToken);
                    if (res.MessageType == WebSocketMessageType.Close) break;

                    if (TryParseBinanceKline(buffer.AsSpan(0, res.Count), out var frame))
                    {
                        _gateway.PublishFrame(frame);
                    }
                }
            }
            catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
            {
                break;
            }
            catch (Exception ex)
            {
                _logger.LogWarning("[BinanceMarketFeed] Live WS unreachable ({Msg}). Running 15s synthetic GBM fallback...", ex.Message);
                using var fallbackCts = CancellationTokenSource.CreateLinkedTokenSource(stoppingToken);
                fallbackCts.CancelAfter(TimeSpan.FromSeconds(15));
                await RunSyntheticGbmFeedAsync(fallbackCts.Token);
            }
        }
    }

    public static bool TryParseBinanceKline(ReadOnlySpan<byte> utf8Json, out LiveStreamFrame frame)
    {
        frame = default;
        try
        {
            using var doc = JsonDocument.Parse(utf8Json.ToArray());
            var root = doc.RootElement;
            var payload = root.TryGetProperty("data", out var dataElem) ? dataElem : root;

            if (!payload.TryGetProperty("k", out var kline))
                return false;

            string symbol = kline.GetProperty("s").GetString() ?? "BTCUSDT";
            string closeStr = kline.GetProperty("c").GetString() ?? "0";
            long eventTime = payload.TryGetProperty("E", out var eProp)
                ? eProp.GetInt64()
                : DateTimeOffset.UtcNow.ToUnixTimeMilliseconds();

            if (!decimal.TryParse(closeStr, NumberStyles.Any, CultureInfo.InvariantCulture, out decimal price) || price <= 0m)
                return false;

            frame = new LiveStreamFrame("MARKET_TICK", symbol, price, null, null, eventTime);
            return true;
        }
        catch
        {
            return false;
        }
    }

    private async Task RunSyntheticGbmFeedAsync(CancellationToken ct)
    {
        var prices = new Dictionary<string, decimal>
        {
            ["BTCUSDT"] = 88500.00m,
            ["ETHUSDT"] = 3150.00m,
            ["SOLUSDT"] = 185.00m
        };
        var rng = new Random();

        while (!ct.IsCancellationRequested)
        {
            foreach (var sym in prices.Keys.ToArray())
            {
                // Box-Muller normal shock: drift + volatility
                double u1 = 1.0 - rng.NextDouble();
                double u2 = 1.0 - rng.NextDouble();
                double z = Math.Sqrt(-2.0 * Math.Log(u1)) * Math.Sin(2.0 * Math.PI * u2);
                decimal pctMove = (decimal)(0.00005 + 0.0012 * z);

                prices[sym] = Math.Round(Math.Max(0.01m, prices[sym] * (1.0m + pctMove)), 4);
                _gateway.PublishFrame(new LiveStreamFrame(
                    "MARKET_TICK",
                    sym,
                    prices[sym],
                    null,
                    "SYNTHETIC_GBM",
                    DateTimeOffset.UtcNow.ToUnixTimeMilliseconds()));
            }
            await Task.Delay(250, ct);
        }
    }
}

using System.Data;
using System.Text.Json;
using Dapper;
using QuantCore.Api.Messaging;
using QuantCore.Api.Services;
using QuantCore.Contracts.Domain;
using QuantCore.Contracts.Messaging;

namespace QuantCore.Api.Endpoints;

public static class AgentEpisodeEndpoints
{
    public static IEndpointRouteBuilder MapAgentEpisodeEndpoints(this IEndpointRouteBuilder app)
    {
        app.MapPost("/api/v1/episodes/commit", async (
            EpisodeCommitRequest req,
            Func<IDbConnection> dbFactory,
            IRabbitMqQueueService mq,
            MarketStreamGateway stream,
            IConfiguration cfg,
            CancellationToken ct) =>
        {
            using var conn = dbFactory();
            conn.Open();
            using var tx = conn.BeginTransaction();

            // 1. Ensure trading_identities record exists
            const string upsertIdentitySql = @"
                INSERT INTO trading_identities (identity_id, actor_type, profile_key)
                VALUES (@IdentityId, @ActorType, @ProfileKey)
                ON CONFLICT (identity_id) DO NOTHING;";
            await conn.ExecuteAsync(new CommandDefinition(
                upsertIdentitySql, new { req.IdentityId, req.ActorType, req.ProfileKey }, tx, cancellationToken: ct));

            // 2. Insert completed episode into trading_history & mark strategy_pool slot COMPLETED
            const string insertHistorySql = @"
                INSERT INTO trading_history
                    (run_id, pool_instance_id, identity_id, profile_key, replica_index, jitter_delta, executed_moves, realized_pnl_usdt, max_drawdown_pct, sharpe_estimate)
                VALUES
                    (@RunId, @PoolInstanceId, @IdentityId, @ProfileKey, @ReplicaIndex, @JitterDeltaJson::jsonb, @MovesJson::jsonb, @RealizedPnlUsdt, @MaxDrawdownPct, @SharpeEstimate);

                UPDATE strategy_pool
                SET status = 'COMPLETED'
                WHERE pool_instance_id = @PoolInstanceId;";

            await conn.ExecuteAsync(new CommandDefinition(
                insertHistorySql,
                new
                {
                    req.RunId,
                    req.PoolInstanceId,
                    req.IdentityId,
                    req.ProfileKey,
                    req.ReplicaIndex,
                    req.JitterDeltaJson,
                    MovesJson = JsonSerializer.Serialize(req.ExecutedMoves),
                    req.RealizedPnlUsdt,
                    req.MaxDrawdownPct,
                    req.SharpeEstimate
                },
                tx,
                cancellationToken: ct));

            // 3. If LiquidateAndReleaseSlices == true, release active strategy_allocations back to portfolio_inventory
            int releasedCount = 0;
            // Always release DB slice reservation at episode boundary so next generation is never starved
            {
                const string releaseSlicesSql = @"
                    WITH released AS (
                        UPDATE strategy_allocations
                        SET status = 'RELEASED'
                        WHERE agent_id = @IdentityId AND status = 'ACTIVE'
                        RETURNING symbol, reserved_quantity
                    ),
                    agg AS (
                        SELECT symbol, SUM(reserved_quantity) AS total_released
                        FROM released
                        GROUP BY symbol
                    )
                    UPDATE portfolio_inventory pi
                    SET allocated_quantity = GREATEST(0, pi.allocated_quantity - agg.total_released),
                        updated_at = NOW()
                    FROM agg
                    WHERE pi.symbol = agg.symbol;";

                releasedCount = await conn.ExecuteAsync(new CommandDefinition(releaseSlicesSql, new { req.IdentityId }, tx, cancellationToken: ct));
            }

            // 4. Check if all K replicas for this profile have finished; if so, trigger evolutionary RECALCULATE_POOL
            const string remainingActiveSql = @"
                SELECT COUNT(1)
                FROM strategy_pool
                WHERE profile_key = @ProfileKey
                  AND status IN ('AVAILABLE', 'CLAIMED');";

            int remaining = await conn.ExecuteScalarAsync<int>(
                new CommandDefinition(remainingActiveSql, new { req.ProfileKey }, tx, cancellationToken: ct));

            tx.Commit();

            bool triggeredRecalc = false;
            if (remaining == 0)
            {
                int kFactor = cfg.GetValue<int>("POOL_REPLICATION_FACTOR", 3);
                mq.SubmitTaskToInputQueue(new InputTaskMessage(
                    Guid.NewGuid(),
                    TaskCommandType.RECALCULATE_POOL,
                    req.ProfileKey,
                    0.75m,
                    kFactor,
                    new[] { "BTCUSDT", "ETHUSDT", "SOLUSDT" },
                    DateTimeOffset.UtcNow
                ));
                triggeredRecalc = true;
            }

            stream.PublishFrame(new LiveStreamFrame(
                "EPISODE_COMMITTED",
                req.ProfileKey,
                req.RealizedPnlUsdt,
                req.IdentityId,
                req.LiquidateAndReleaseSlices ? "LIQUIDATED" : "CARRIED",
                DateTimeOffset.UtcNow.ToUnixTimeMilliseconds()));

            return Results.Ok(new EpisodeCommitResponse(true, req.RunId, releasedCount, triggeredRecalc));
        });

        app.MapPost("/api/v1/episodes/reset", (EpisodeResetCommand cmd, MarketStreamGateway stream) =>
        {
            stream.PublishFrame(new LiveStreamFrame(
                "FORCE_EPISODE_RESET",
                cmd.ProfileKey ?? "ALL",
                0m,
                null,
                cmd.Reason,
                DateTimeOffset.UtcNow.ToUnixTimeMilliseconds()));
            return Results.Ok(new { broadcasted = true, cmd.Reason });
        });


        app.MapGet("/api/v1/episodes/summary", async (Func<IDbConnection> dbFactory, CancellationToken ct) =>
        {
            using var conn = dbFactory();
            conn.Open();

            const string kpiSql = @"
                SELECT
                    COUNT(1) AS total_episodes,
                    COALESCE(SUM(realized_pnl_usdt), 0) AS total_pnl_usdt,
                    COALESCE(AVG(sharpe_estimate), 0) AS avg_sharpe,
                    COALESCE(MAX(max_drawdown_pct), 0) AS worst_drawdown_pct,
                    (SELECT COUNT(1) FROM nn_model_checkpoints) AS total_checkpoints
                FROM trading_history;";

            var kpis = await conn.QuerySingleAsync(new CommandDefinition(kpiSql, cancellationToken: ct));

            const string recentEpisodesSql = @"
                SELECT run_id, profile_key, replica_index, realized_pnl_usdt, max_drawdown_pct, sharpe_estimate, completed_at
                FROM trading_history
                ORDER BY completed_at DESC
                LIMIT 15;";

            var recentEpisodes = await conn.QueryAsync(new CommandDefinition(recentEpisodesSql, cancellationToken: ct));

            const string nnCheckpointsSql = @"
                SELECT model_id, version_tag, confidence_score, is_promoted_live, created_at
                FROM nn_model_checkpoints
                ORDER BY created_at DESC
                LIMIT 8;";

            var checkpoints = await conn.QueryAsync(new CommandDefinition(nnCheckpointsSql, cancellationToken: ct));

            const string poolStateSql = @"
                SELECT profile_key, replica_index, status, risk_dial_r, strategy_payload::text AS strategy_payload
                FROM strategy_pool
                WHERE status IN ('AVAILABLE', 'CLAIMED')
                ORDER BY profile_key, replica_index;";

            var poolState = await conn.QueryAsync(new CommandDefinition(poolStateSql, cancellationToken: ct));

            return Results.Ok(new
            {
                kpis,
                recent_episodes = recentEpisodes,
                nn_checkpoints = checkpoints,
                pool_state = poolState
            });
        });

        return app;
    }
